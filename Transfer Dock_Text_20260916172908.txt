function $(id){
 return document.getElementById(id);
}

function save(){
 try{
  localStorage.setItem("creatoraddain_state",JSON.stringify({
   user:state.user,
   gamesPlayed:state.gamesPlayed,
   winStreak:state.winStreak,
   history:state.history
  }));
 }catch(e){}
}

function purgeLegacyDemo(){
 try{
  localStorage.removeItem("creatoraddain_state");
 }catch(e){}
}

function load(){
 try{
  const x=JSON.parse(localStorage.getItem("creatoraddain_state"));
  if(x){
   delete x.balance;
   Object.assign(state,x);
  }
 }catch(e){}

 updateUI();
 renderGames();
}

function updateUI(){
 $("balance").textContent = state.user ? "0.00" : "--";
 $("gamesPlayed").textContent=state.gamesPlayed;
 $("winStreak").textContent=state.winStreak;

 if(state.user){
  $("authBtn").textContent="LOGOUT";
  $("profileName").textContent=state.user.name;
  $("avatar").textContent=(state.user.name||"P").charAt(0).toUpperCase();
 }else{
  $("authBtn").textContent="LOGIN";
  $("profileName").textContent="Guest Player";
  $("avatar").textContent="👤";
 }

 renderHistory();
}

function showMessage(msg){
 const t=$("toast");
 t.textContent=msg;
 t.classList.add("show");

 setTimeout(()=>{
  t.classList.remove("show");
 },2200);
}

function go(id){
 const el=$(id);
 if(el)el.scrollIntoView({behavior:"smooth"});
}

function openAuth(){
 if(state.user){
  state.user=null;
  save();
  updateUI();
  showMessage("Logged out.");
  return;
 }

 $("authModal").classList.add("show");
}

function closeAuth(){
 $("authModal").classList.remove("show");
}

function demoLogin(){
 const name=$("username").value.trim()||"Player";

 state.user={
  name:name,
  level:1
 };

 save();
 updateUI();
 closeAuth();
 showMessage("Login successful.");
}

function renderGames(){
 $("gamesGrid").innerHTML=games.map(g=>`
  <div class="game-card">
   <div class="game-icon">${g.icon}</div>
   <h3>${g.name}</h3>
   <p>${g.desc}</p>
   <span class="badge">GAME ${g.id}</span>
   <button class="btn" onclick="startGame(${g.id})">
    PLAY NOW
   </button>
  </div>
 `).join("");
}

function startGame(id){
 const g=games.find(x=>x.id===id);

 if(!g)return;

 if(!state.user){
  showMessage("Pehle LOGIN karo.");
  openAuth();
  return;
 }

 state.currentGame=g;
 $("playTitle").textContent=g.icon+" "+g.name;
 $("playScreen").classList.add("show");

 renderGame(g.fn);
}

function exitGame(){
 $("playScreen").classList.remove("show");
 $("gameArea").innerHTML="";
 state.currentGame=null;
}

function finishGame(result,reason){
 state.gamesPlayed++;

 if(result==="win"){
  state.winStreak++;
 }else{
  state.winStreak=0;
 }

 state.history.unshift({
  game:state.currentGame?state.currentGame.name:"Game",
  result:result,
  reason:reason,
  time:new Date().toLocaleString()
 });

 if(state.history.length>50){
  state.history.length=50;
 }

 save();
 updateUI();
}

function renderHistory(){
 const box=$("historyList");

 if(!state.history.length){
  box.innerHTML='<p class="muted">No games played yet.</p>';
  return;
 }

 box.innerHTML=state.history.map(h=>`
  <div class="history-row">
   <span>${h.game}</span>
   <span>${h.result==="win"?"✅":"❌"}</span>
  </div>
 `).join("");
}

function simpleGame(icon,name){
 $("gameArea").innerHTML=`
 <div class="game-title">
  <h2>${icon} ${name}</h2>
  <p class="muted">Game Mode Ready</p>
 </div>
 <div class="center">
  <button class="btn" onclick="finishGame('win','Completed ${name}')">Complete Game</button>
 </div>
 `;
}

function gameWheel(){
 let spinning=false;

 $("gameArea").innerHTML=`
 <div class="game-title">
  <h2>🎡 Lucky Wheel</h2>
  <p class="muted">Spin the wheel.</p>
 </div>

 <div class="wheel" id="wheel">
  <span id="wheelText">SPIN</span>
 </div>

 <div class="center">
  <p id="wheelStatus" class="muted">Ready</p>
  <button class="btn" id="wheelBtn">🎡 SPIN NOW</button>
 </div>
 `;

 $("wheelBtn").onclick=()=>{
  if(spinning)return;

  spinning=true;
  $("wheelBtn").disabled=true;

  const prizes=["1×","2×","5×","10×"];
  const chosen=prizes[Math.floor(Math.random()*4)];
  const index=prizes.indexOf(chosen);
  const rotation=1800+(index*90)+45;

  $("wheel").style.transform="rotate("+rotation+"deg)";
  $("wheelStatus").textContent="Spinning...";

  setTimeout(()=>{
   $("wheelText").textContent=chosen;
   $("wheelStatus").textContent="Result: "+chosen;

   finishGame("win","Lucky Wheel: "+chosen);
   showMessage("Wheel result: "+chosen);

   spinning=false;
   $("wheelBtn").disabled=false;
  },3100);
 };
}

function gameQuiz(){
 const questions=[
  ["India ki capital kya hai?",["Mumbai","New Delhi","Kolkata","Chennai"],1],
  ["7 × 8 = ?",["54","56","64","48"],1],
  ["Water ka chemical formula?",["CO2","O2","H2O","NaCl"],2],
  ["Sun ek kya hai?",["Planet","Star","Moon","Comet"],1],
  ["Earth kis galaxy mein hai?",["Andromeda","Milky Way","Whirlpool","Sombrero"],1]
 ];

 let index=0;
 let score=0;

 function draw(){
  const q=questions[index];

  $("gameArea").innerHTML=`
   <div class="game-title">
    <h2>🧠 Question Master</h2>
    <p class="muted">Question ${index+1} / ${questions.length}</p>
   </div>

   <div class="result">
    <div class="quiz-q">${q[0]}</div>
   </div>

   <div class="choice-grid">
    ${q[1].map((x,i)=>`
     <button class="choice" data-answer="${i}">${x}</button>
    `).join("")}
   </div>
  `;

  document.querySelectorAll("[data-answer]").forEach(btn=>{
   btn.onclick=()=>{
    const answer=Number(btn.dataset.answer);

    if(answer===q[2]){
     score++;
     showMessage("Correct!");
    }else{
     showMessage("Wrong answer.");
    }

    index++;

    if(index>=questions.length){
     const win=score>=3;
     finishGame(win?"win":"loss","Quiz score: "+score+"/"+questions.length);
     $("gameArea").innerHTML=`
      <div class="center">
       <div class="big-number">${win?"🏆":"📚"}</div>
       <h2>Quiz Finished</h2>
      </div>
     `;
    }else{
     draw();
    }
   };
  });
 }

 draw();
}

function renderGame(fn){
 if(fn==="wheel")return gameWheel();
 if(fn==="quiz")return gameQuiz();

}
